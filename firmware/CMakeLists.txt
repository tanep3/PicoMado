cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH)
  if(DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
  else()
    message(FATAL_ERROR "Set PICO_SDK_PATH to your installed Pico SDK")
  endif()
endif()
# The local SDK requests picotool 2.3, while raspi5 has 2.1.1. RP2040
# conversion is supported by the installed tool; do not fetch or install.
set(PICO_NO_PICOTOOL 1)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(picomado_firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

set(PM_DEVICE "" CACHE STRING "Explicit device name for this build")
if(NOT PM_DEVICE MATCHES "^[A-Za-z0-9_-]+$")
  message(FATAL_ERROR "Set -DPM_DEVICE to a device directory name")
endif()
get_filename_component(PM_BUILD_BASENAME "${CMAKE_BINARY_DIR}" NAME)
if(NOT PM_BUILD_BASENAME STREQUAL PM_DEVICE)
  message(FATAL_ERROR "Build directory must end with the PM_DEVICE name")
endif()
option(PM_USE_SAMPLE "Build a dummy sample device; never flash this UF2" OFF)
option(PM_PREPARE_UNBOUND "Compile before the physical board is identified; cannot flash" OFF)
option(PM_PREPARE_UNBOUND_PANEL "Build a V4 display-enabled candidate without changing device config" OFF)
if(PM_PREPARE_UNBOUND_PANEL AND (NOT PM_PREPARE_UNBOUND OR PM_USE_SAMPLE))
  message(FATAL_ERROR "Panel candidate requires --unbound-panel")
endif()
if(PM_USE_SAMPLE)
  set(PM_EXPECTED_BUILD_DIR "${CMAKE_CURRENT_LIST_DIR}/build/.samples/${PM_DEVICE}")
elseif(PM_PREPARE_UNBOUND_PANEL)
  set(PM_EXPECTED_BUILD_DIR "${CMAKE_CURRENT_LIST_DIR}/build/.unbound-panel/${PM_DEVICE}")
elseif(PM_PREPARE_UNBOUND)
  set(PM_EXPECTED_BUILD_DIR "${CMAKE_CURRENT_LIST_DIR}/build/.unbound/${PM_DEVICE}")
else()
  set(PM_EXPECTED_BUILD_DIR "${CMAKE_CURRENT_LIST_DIR}/build/${PM_DEVICE}")
endif()
if(NOT CMAKE_BINARY_DIR STREQUAL PM_EXPECTED_BUILD_DIR)
  message(FATAL_ERROR "Use the device-specific build directory: ${PM_EXPECTED_BUILD_DIR}")
endif()
set(PM_DEVICE_DIR "${CMAKE_CURRENT_LIST_DIR}/devices/${PM_DEVICE}")
file(READ "${PM_DEVICE_DIR}/profile.txt" PM_PROFILE)
string(STRIP "${PM_PROFILE}" PM_PROFILE)
if(NOT PM_PROFILE MATCHES "^(pico_w_lcd_160x80_4|pico_w_eink_250x122_[02])$")
  message(FATAL_ERROR "Unverified hardware profile")
endif()
set(PM_PROFILE_HEADER "${CMAKE_CURRENT_LIST_DIR}/profiles/${PM_PROFILE}.h")
file(READ "${CMAKE_CURRENT_LIST_DIR}/profiles/${PM_PROFILE}.board" PM_PROFILE_BOARD)
string(STRIP "${PM_PROFILE_BOARD}" PM_PROFILE_BOARD)
if(NOT PM_PROFILE_BOARD STREQUAL "pico_w" OR NOT PICO_BOARD STREQUAL PM_PROFILE_BOARD)
  message(FATAL_ERROR "Profile board does not match PICO_BOARD")
endif()

if(PM_USE_SAMPLE)
  set(PM_CONFIG_FILE "${PM_DEVICE_DIR}/config.sample.h")
  set(PM_BOARD_FILE "${PM_DEVICE_DIR}/board-id.sample.txt")
else()
  set(PM_CONFIG_FILE "${PM_DEVICE_DIR}/config.h")
  set(PM_BOARD_FILE "${PM_DEVICE_DIR}/board-id.txt")
endif()
if(NOT EXISTS "${PM_CONFIG_FILE}" OR (NOT PM_PREPARE_UNBOUND AND NOT EXISTS "${PM_BOARD_FILE}"))
  message(FATAL_ERROR "Device config or board ID file is missing in ${PM_DEVICE_DIR}")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tools/validate_hardware_profile.py"
  --config "${PM_CONFIG_FILE}" --profile "${PM_PROFILE_HEADER}" --compiler "${CMAKE_C_COMPILER}"
  RESULT_VARIABLE PM_PROFILE_VALID OUTPUT_VARIABLE PM_PROFILE_STYLE ERROR_QUIET)
if(NOT PM_PROFILE_VALID EQUAL 0)
  message(FATAL_ERROR "Device hardware differs from the selected verified profile")
endif()
string(STRIP "${PM_PROFILE_STYLE}" PM_PROFILE_STYLE)
if(NOT PM_PROFILE_STYLE STREQUAL "profile" AND NOT PM_PROFILE_STYLE STREQUAL "legacy")
  message(FATAL_ERROR "Unknown hardware configuration style")
endif()
file(STRINGS "${PM_CONFIG_FILE}" PM_ID_LINES REGEX "^[ \t]*#define[ \t]+PM_DEVICE_ID[ \t]+")
list(LENGTH PM_ID_LINES PM_ID_LINE_COUNT)
if(NOT PM_ID_LINE_COUNT EQUAL 1)
  message(FATAL_ERROR "Exactly one PM_DEVICE_ID definition is required")
endif()
string(REGEX MATCH "^[ \t]*#define[ \t]+PM_DEVICE_ID[ \t]+\"([A-Za-z0-9_-]+)\"[ \t]*$"
  PM_ID_MATCH "${PM_ID_LINES}")
if(NOT PM_ID_MATCH OR NOT CMAKE_MATCH_1 STREQUAL PM_DEVICE)
  message(FATAL_ERROR "Config PM_DEVICE_ID does not match the explicit PM_DEVICE build selection")
endif()
if(PM_PREPARE_UNBOUND)
  set(PM_BOARD_ID "0000000000000000")
else()
  file(READ "${PM_BOARD_FILE}" PM_BOARD_ID)
endif()
string(STRIP "${PM_BOARD_ID}" PM_BOARD_ID)
string(LENGTH "${PM_BOARD_ID}" PM_BOARD_ID_LENGTH)
if(NOT PM_BOARD_ID_LENGTH EQUAL 16 OR NOT PM_BOARD_ID MATCHES "^[0-9A-Fa-f]+$")
  message(FATAL_ERROR "The device board ID must be 16 hexadecimal digits")
endif()
if(PM_USE_SAMPLE)
  set(PM_SAMPLE_DEFINE PM_USE_SAMPLE_CONFIG=1)
endif()

if(PM_PREPARE_UNBOUND_PANEL)
  file(READ "${PM_CONFIG_FILE}" PM_PANEL_CONFIG)
  if(NOT PM_PROFILE MATCHES "^pico_w_eink_" OR
     NOT PM_PANEL_CONFIG MATCHES "#define[ \t]+PM_EINK_PANEL_CONNECTED[ \t]+[01]")
    message(FATAL_ERROR "Display-enabled candidate supports the configured V4 E-ink only")
  endif()
  string(REGEX REPLACE "(#define[ \t]+PM_EINK_PANEL_CONNECTED[ \t]+)[01]" "\\11" PM_PANEL_CONFIG "${PM_PANEL_CONFIG}")
  set(PM_DEVICE_DIR "${CMAKE_BINARY_DIR}/panel-config")
  file(MAKE_DIRECTORY "${PM_DEVICE_DIR}")
  file(WRITE "${PM_DEVICE_DIR}/config.h" "${PM_PANEL_CONFIG}")
endif()

option(PM_ACK_PATH_PROBE "One bounded ACK/input transaction; off by default" OFF)
option(PM_USB_BUTTON_SIM "Bounded explicit USB virtual input; off by default" OFF)
option(PM_TX_LINK_PROBE "Bounded target linkoutput observations; off by default" OFF)
if(PM_TX_LINK_PROBE AND NOT PM_ACK_PATH_PROBE)
  message(FATAL_ERROR "TX link probe requires ACK path probe run ownership")
endif()
option(PM_CYW43_INNER_PROBE "Bounded CYW43 credit/RX/pre-bus observation; off by default" OFF)
option(PM_CYW43_SPI_REAL_SIZE "LCD-only private LL comparison of upstream SPI command length fix" OFF)
if(PM_CYW43_SPI_REAL_SIZE AND (NOT PM_CYW43_INNER_PROBE OR NOT PM_DEVICE STREQUAL "PicoMado-lcd1"))
  message(FATAL_ERROR "SPI real-size comparison requires the LCD private inner-probe build")
endif()
if(PM_CYW43_INNER_PROBE)
  if(NOT PM_TX_LINK_PROBE OR NOT PM_ACK_PATH_PROBE)
    message(FATAL_ERROR "CYW43 inner probe requires TX/AP2 run ownership")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(PM_PRIVATE_LL "${CMAKE_CURRENT_BINARY_DIR}/private-cyw43/cyw43_ll.c")
  if(PM_CYW43_SPI_REAL_SIZE)
    set(PM_SPI_REAL_SIZE_ARG --spi-real-size)
  else()
    set(PM_SPI_REAL_SIZE_ARG "")
  endif()
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/tools/prepare_cyw43_inner.py
    --source ${PICO_CYW43_DRIVER_PATH}/src/cyw43_ll.c --output ${PM_PRIVATE_LL} ${PM_SPI_REAL_SIZE_ARG}
    RESULT_VARIABLE PM_LL_GENERATION)
  if(NOT PM_LL_GENERATION EQUAL 0)
    message(FATAL_ERROR "Pinned private CYW43 diagnostic source generation failed")
  endif()
  get_target_property(PM_DRIVER_SOURCES cyw43_driver INTERFACE_SOURCES)
  set(PM_REPLACED_LL 0)
  set(PM_NEW_DRIVER_SOURCES "")
  foreach(PM_SOURCE IN LISTS PM_DRIVER_SOURCES)
    if(PM_SOURCE STREQUAL "${PICO_CYW43_DRIVER_PATH}/src/cyw43_ll.c")
      list(APPEND PM_NEW_DRIVER_SOURCES "${PM_PRIVATE_LL}")
      math(EXPR PM_REPLACED_LL "${PM_REPLACED_LL} + 1")
    else()
      list(APPEND PM_NEW_DRIVER_SOURCES "${PM_SOURCE}")
    endif()
  endforeach()
  if(NOT PM_REPLACED_LL EQUAL 1)
    message(FATAL_ERROR "Expected exactly one CYW43 LL translation unit")
  endif()
  set_property(TARGET cyw43_driver PROPERTY INTERFACE_SOURCES "${PM_NEW_DRIVER_SOURCES}")
endif()
option(PM_EVENT_STREAM "Generic bounded boundary events over the existing USB owner" OFF)
option(PM_STREAM_DIAGNOSTICS "Bounded stream diagnostics; never changes playback policy" OFF)
add_library(picomado_core STATIC src/core.c src/button_sim.c src/video.c src/endpoint_policy.c src/eex_wire.c src/eex_exchange.c src/endpoint_client.c)
target_compile_options(picomado_core PRIVATE -fstack-usage)
if(PM_PROFILE_STYLE STREQUAL "profile")
  # Absolute tracked path: no quoted-include lookup in the device directory.
  target_compile_options(picomado_core PUBLIC -include "${PM_PROFILE_HEADER}")
endif()
target_include_directories(picomado_core PUBLIC ${PM_DEVICE_DIR} ${CMAKE_CURRENT_LIST_DIR}/profiles ${CMAKE_CURRENT_LIST_DIR}/src)
target_compile_definitions(picomado_core PUBLIC ${PM_SAMPLE_DEFINE} PM_STREAM_DIAGNOSTICS=$<BOOL:${PM_STREAM_DIAGNOSTICS}>)

file(STRINGS "${PM_PROFILE_HEADER}" PM_DISPLAY_LINES REGEX "^[ \t]*#define[ \t]+PM_DISPLAY_TYPE[ \t]+")
if(PM_DISPLAY_LINES MATCHES "PM_DISPLAY_EINK")
  set(PM_DISPLAY_SOURCES src/display_eink.c)
  set(PM_APP_TARGET picomado_eink)
else()
  set(PM_DISPLAY_SOURCES src/display.c src/lcd.c)
  set(PM_APP_TARGET picomado_lcd)
endif()

if(NOT PM_USE_SAMPLE AND PM_APP_TARGET STREQUAL "picomado_lcd")
  add_executable(picomado_selftest src/selftest.c src/display.c src/lcd.c)
  target_include_directories(picomado_selftest PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
  target_link_libraries(picomado_selftest pico_stdlib hardware_spi picomado_core)
  pico_enable_stdio_usb(picomado_selftest 1)
  pico_enable_stdio_uart(picomado_selftest 0)
  pico_add_extra_outputs(picomado_selftest)
  add_custom_command(TARGET picomado_selftest POST_BUILD
    COMMAND picotool uf2 convert $<TARGET_FILE:picomado_selftest> -t elf
      ${CMAKE_CURRENT_BINARY_DIR}/${PM_DEVICE}-selftest.uf2 -t uf2)

  add_executable(picomado_backlight_polarity diagnostics/backlight-polarity/main.c src/lcd.c)
  target_include_directories(picomado_backlight_polarity PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/src)
  target_compile_definitions(picomado_backlight_polarity PRIVATE
    PM_LCD_FRAME_CS_HOLD=1 PM_LCD_CONTROL_SPI_BAUD=PM_LCD_SPI_BAUD PM_LCD_CAPTURE_ENABLE=0
    PM_EXPECTED_BOARD_ID="${PM_BOARD_ID}" PICO_STACK_SIZE=0x1000)
  target_link_libraries(picomado_backlight_polarity pico_stdlib pico_unique_id hardware_spi
    pico_cyw43_arch_lwip_poll picomado_core)
  pico_enable_stdio_usb(picomado_backlight_polarity 1)
  pico_enable_stdio_uart(picomado_backlight_polarity 0)
  pico_add_extra_outputs(picomado_backlight_polarity)
  add_custom_command(TARGET picomado_backlight_polarity POST_BUILD
    COMMAND picotool uf2 convert $<TARGET_FILE:picomado_backlight_polarity> -t elf
      ${CMAKE_CURRENT_BINARY_DIR}/${PM_DEVICE}-backlight-polarity.uf2 -t uf2)
endif()

if(NOT PM_USE_SAMPLE AND NOT PM_PREPARE_UNBOUND)
  set(PM_AWAY_EEX1_DEFAULT ON)
else()
  set(PM_AWAY_EEX1_DEFAULT OFF)
endif()
option(PM_AWAY_EEX1 "Resolve AWAY SOCKS endpoint using foreground Android EEX1" ${PM_AWAY_EEX1_DEFAULT})
if(PM_AWAY_EEX1_DEFAULT AND NOT PM_AWAY_EEX1)
  message(FATAL_ERROR "Production AWAY requires EEX1; PM_AWAY_EEX1=OFF is not a flashable product build")
endif()
set(PM_EEX1_PORT 48116 CACHE STRING "EEX1 UDP discovery port")
if(NOT PM_EEX1_PORT MATCHES "^[1-9][0-9]*$" OR PM_EEX1_PORT GREATER 65535)
  message(FATAL_ERROR "PM_EEX1_PORT must be1..65535")
endif()
add_executable(${PM_APP_TARGET} src/cyw43_inner_probe.c src/usb_control.c src/tx_link_probe.c src/ack_probe.c src/stream_diag.c src/endpoint_udp_sdk.c src/main.c ${PM_DISPLAY_SOURCES} src/net.c src/ws_accept.c src/trace.c src/trace_usb.c src/rx_stats.c)
target_include_directories(${PM_APP_TARGET} PRIVATE ${PM_DEVICE_DIR} ${CMAKE_CURRENT_LIST_DIR}/profiles ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/src)
target_compile_options(${PM_APP_TARGET} PRIVATE -fstack-usage)
target_compile_definitions(${PM_APP_TARGET} PRIVATE PM_EVENT_STREAM=$<BOOL:${PM_EVENT_STREAM}> PM_CYW43_INNER_PROBE=$<BOOL:${PM_CYW43_INNER_PROBE}> PM_USB_BUTTON_SIM=$<BOOL:${PM_USB_BUTTON_SIM}> PM_TX_LINK_PROBE=$<BOOL:${PM_TX_LINK_PROBE}> PM_ACK_PATH_PROBE=$<BOOL:${PM_ACK_PATH_PROBE}> PM_AWAY_EEX1=$<BOOL:${PM_AWAY_EEX1}> PM_EEX1_PORT=${PM_EEX1_PORT} PM_NETWORK_BUILD=1 PM_NORMAL_BUILD=1 PM_TRACE_ENABLE=1 ${PM_SAMPLE_DEFINE}
  PM_LCD_FRAME_CS_HOLD=1 PM_LCD_CONTROL_SPI_BAUD=PM_LCD_SPI_BAUD
  PM_LCD_CAPTURE_ENABLE=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 PICO_STDIO_DEADLOCK_TIMEOUT_MS=0
  PM_EXPECTED_BOARD_ID="${PM_BOARD_ID}" PM_BUILD_DEVICE="${PM_DEVICE}"
  PICO_STACK_SIZE=0x1000)
if(PM_APP_TARGET STREQUAL "picomado_lcd")
  target_compile_definitions(${PM_APP_TARGET} PRIVATE PM_TRACE_STATS=1 PM_TRACE_VERBOSE=0)
endif()
target_link_libraries(${PM_APP_TARGET} pico_stdlib pico_unique_id pico_rand hardware_spi hardware_flash pico_cyw43_arch_lwip_threadsafe_background picomado_core)
pico_enable_stdio_usb(${PM_APP_TARGET} 1)
pico_enable_stdio_uart(${PM_APP_TARGET} 0)
pico_add_extra_outputs(${PM_APP_TARGET})
add_custom_command(TARGET ${PM_APP_TARGET} POST_BUILD
  COMMAND picotool uf2 convert $<TARGET_FILE:${PM_APP_TARGET}> -t elf
    ${CMAKE_CURRENT_BINARY_DIR}/${PM_DEVICE}.uf2 -t uf2)
