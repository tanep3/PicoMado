cmake_minimum_required(VERSION 3.13)
# Select an installed SDK explicitly. No SDK/tool downloads.
if(NOT DEFINED PICO_SDK_PATH)
  set(PICO_SDK_PATH /home/tane/pico/pico-sdk)
endif()
set(PICO_NO_PICOTOOL 1)
option(PM_REPRO_YOUTUBE_CYW43 "Use the working YouTube CYW43 driver snapshot" ON)
if(PM_REPRO_YOUTUBE_CYW43)
  set(PICO_CYW43_DRIVER_PATH /home/tane/dev/PicoW/PicoW_Youtube_Player/diagnostics/pico/cyw43-driver)
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(picomado_lcd_repro C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
get_filename_component(PM_FIRMWARE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PM_REPRO_DEVICE "" CACHE STRING "Existing device directory, explicitly selected")
set(PM_REPRO_SPI_BAUD 5000000 CACHE STRING "Initial uniform control/frame SPI request")
option(PM_REPRO_WIFI_HOME "Join only the existing configured home network" OFF)
option(PM_REPRO_SAMPLE "Compile dummy config; never flash this build" ON)
option(PM_REPRO_CAPTURE "Snapshot frames 1/2/7 before LCD SPI; export on USB D request" OFF)
option(PM_REPRO_HOLD_FRAME_CS "Hold CS across CASET/RASET/RAMWR/pixels at equal rates" OFF)
if(NOT PM_REPRO_DEVICE MATCHES "^[A-Za-z0-9_-]+$")
  message(FATAL_ERROR "Set PM_REPRO_DEVICE to an existing device directory")
endif()
if(NOT PM_REPRO_SPI_BAUD MATCHES "^(1000000|5000000|20000000)$")
  message(FATAL_ERROR "Reproduction rates are 1, 5 or 20 MHz")
endif()
if(PM_REPRO_SAMPLE AND PM_REPRO_WIFI_HOME)
  message(FATAL_ERROR "Dummy sample must not attempt a Wi-Fi join")
endif()
set(PM_DEVICE_DIR "${PM_FIRMWARE_ROOT}/devices/${PM_REPRO_DEVICE}")
if(PM_REPRO_SAMPLE)
  set(PM_CONFIG_FILE "${PM_DEVICE_DIR}/config.sample.h")
  set(PM_BOARD_FILE "${PM_DEVICE_DIR}/board-id.sample.txt")
else()
  set(PM_CONFIG_FILE "${PM_DEVICE_DIR}/config.h")
  set(PM_BOARD_FILE "${PM_DEVICE_DIR}/board-id.txt")
endif()
if(NOT EXISTS "${PM_CONFIG_FILE}" OR NOT EXISTS "${PM_BOARD_FILE}")
  message(FATAL_ERROR "Selected device config/board ID is missing")
endif()
file(STRINGS "${PM_CONFIG_FILE}" PM_ID_LINE REGEX "^[ \t]*#define[ \t]+PM_DEVICE_ID[ \t]+")
string(REGEX MATCH "PM_DEVICE_ID[ \t]+\"([A-Za-z0-9_-]+)\"" PM_ID_MATCH "${PM_ID_LINE}")
if(NOT PM_ID_MATCH OR NOT CMAKE_MATCH_1 STREQUAL PM_REPRO_DEVICE)
  message(FATAL_ERROR "Config device ID does not match the selected reproduction device")
endif()
file(READ "${PM_BOARD_FILE}" PM_BOARD_ID)
string(STRIP "${PM_BOARD_ID}" PM_BOARD_ID)
string(LENGTH "${PM_BOARD_ID}" PM_BOARD_LENGTH)
if(NOT PM_BOARD_LENGTH EQUAL 16 OR NOT PM_BOARD_ID MATCHES "^[0-9A-Fa-f]+$")
  message(FATAL_ERROR "Invalid board ID")
endif()
add_executable(picomado_lcd_repro
  main.c "${PM_FIRMWARE_ROOT}/src/lcd.c" "${PM_FIRMWARE_ROOT}/src/core.c")
if(PM_REPRO_CAPTURE)
  target_sources(picomado_lcd_repro PRIVATE capture.c)
  target_compile_definitions(picomado_lcd_repro PRIVATE PM_LCD_CAPTURE_ENABLE=1)
endif()
if(PM_REPRO_HOLD_FRAME_CS)
  target_compile_definitions(picomado_lcd_repro PRIVATE PM_LCD_FRAME_CS_HOLD=1)
endif()
target_include_directories(picomado_lcd_repro PRIVATE
  "${CMAKE_CURRENT_LIST_DIR}" "${PM_DEVICE_DIR}" "${PM_FIRMWARE_ROOT}/src"
  /home/tane/dev/PicoW/PicoW_Youtube_Player/diagnostics/pico)
target_compile_definitions(picomado_lcd_repro PRIVATE
  PM_REPRO_SPI_BAUD=${PM_REPRO_SPI_BAUD}
  PM_REPRO_WIFI_HOME=$<BOOL:${PM_REPRO_WIFI_HOME}>
  PM_REPRO_SAMPLE=$<BOOL:${PM_REPRO_SAMPLE}>
  PM_REPRO_BOARD_ID="${PM_BOARD_ID}" PICO_STACK_SIZE=0x1000)
if(PM_REPRO_SAMPLE)
  target_compile_definitions(picomado_lcd_repro PRIVATE PM_USE_SAMPLE_CONFIG=1)
endif()
if(PM_REPRO_YOUTUBE_CYW43)
  target_sources(picomado_lcd_repro PRIVATE
    /home/tane/dev/PicoW/PicoW_Youtube_Player/diagnostics/pico/cyw43_diag.c)
  target_compile_definitions(picomado_lcd_repro PRIVATE PLAYER_CYW43_DIAG=1)
endif()
target_link_libraries(picomado_lcd_repro pico_stdlib pico_unique_id hardware_spi
  pico_cyw43_arch_lwip_threadsafe_background)
pico_enable_stdio_usb(picomado_lcd_repro 1)
pico_enable_stdio_uart(picomado_lcd_repro 0)
pico_add_extra_outputs(picomado_lcd_repro)
add_custom_command(TARGET picomado_lcd_repro POST_BUILD
  COMMAND picotool uf2 convert $<TARGET_FILE:picomado_lcd_repro> -t elf
    "${CMAKE_CURRENT_BINARY_DIR}/picomado_lcd_repro.uf2" -t uf2)
